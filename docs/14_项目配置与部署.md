# 14 - 项目配置与部署

## 1. 配置管理（Viper）

### 安装

```bash
go get github.com/spf13/viper
```

### config.yaml

```yaml
server:
  port: 8080
  mode: debug  # debug / release / test

database:
  host: 127.0.0.1
  port: 3306
  username: root
  password: password
  dbname: go_demo
  charset: utf8mb4
```

### config/config.go

```go
package config

import (
    "fmt"
    "github.com/spf13/viper"
)

type Config struct {
    Server   ServerConfig
    Database DatabaseConfig
}

type ServerConfig struct {
    Port string
    Mode string
}

type DatabaseConfig struct {
    Host     string
    Port     int
    Username string
    Password string
    Dbname   string
    Charset  string
}

func Load() *Config {
    viper.SetConfigName("config")
    viper.SetConfigType("yaml")
    viper.AddConfigPath(".")           // 当前目录
    viper.AddConfigPath("./config")

    // 支持环境变量覆盖
    viper.AutomaticEnv()

    if err := viper.ReadInConfig(); err != nil {
        panic(fmt.Errorf("读取配置失败: %w", err))
    }

    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        panic(err)
    }
    return &cfg
}

func (c *Config) GetDSN() string {
    return fmt.Sprintf("%s:%s@tcp(%s:%d)/%s?charset=%s&parseTime=True&loc=Local",
        c.Database.Username,
        c.Database.Password,
        c.Database.Host,
        c.Database.Port,
        c.Database.Dbname,
        c.Database.Charset,
    )
}
```

### 使用

```go
func main() {
    cfg := config.Load()
    fmt.Println("端口:", cfg.Server.Port)
    fmt.Println("数据库:", cfg.Database.Host)
}
```

## 3. 日志（Logrus / Zap）

### 安装 Zap

```bash
go get go.uber.org/zap
```

### 使用

```go
package logger

import (
    "go.uber.org/zap"
)

var Log *zap.Logger

func Init() {
    var err error
    Log, err = zap.NewProduction()
    if err != nil {
        panic(err)
    }
}

func Sync() {
    _ = Log.Sync()
}
```

```go
logger.Log.Info("用户登录",
    zap.String("username", "tom"),
    zap.Int("user_id", 1),
)
```

### Gin 中间件用 Zap

```go
func ZapLogger(logger *zap.Logger) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()
        c.Next()

        logger.Info("request",
            zap.String("method", c.Request.Method),
            zap.String("path", c.Request.URL.Path),
            zap.Int("status", c.Writer.Status()),
            zap.Duration("duration", time.Since(start)),
        )
    }
}

r.Use(ZapLogger(logger.Log))
```

## 5. 编译与构建

### 基本编译

```bash
# 编译当前项目
go build -o myapp main.go

# 运行
./myapp        # Linux/Mac
myapp.exe      # Windows
```

### 交叉编译

```bash
# 编译成 Linux 版本（在 Windows 上交叉编译）
set GOOS=linux
set GOARCH=amd64
go build -o myapp main.go

# 一行命令
GOOS=linux GOARCH=amd64 go build -o myapp main.go
```

常见目标：
- Linux 64位：`GOOS=linux GOARCH=amd64`
- Windows 64位：`GOOS=windows GOARCH=amd64`
- Mac（Apple Silicon）：`GOOS=darwin GOARCH=arm64`

### 减小二进制体积

```bash
# 去掉调试信息
go build -ldflags="-s -w" -o myapp main.go
```

### 版本信息嵌入

```go
// main.go
var (
    Version   = "dev"
    BuildTime = "unknown"
)

func main() {
    fmt.Printf("Version: %s, Built at: %s\n", Version, BuildTime)
}
```

```bash
go build -ldflags="-X 'main.Version=v1.0.0' -X 'main.BuildTime=$(date)'" -o myapp main.go
```

## 6. 部署到 Linux 服务器

### 步骤

1. **本地编译**（交叉编译为 Linux 版本）
   ```bash
   GOOS=linux GOARCH=amd64 go build -o myapp main.go
   ```

2. **上传到服务器**
   ```bash
   scp myapp config.yaml user@server:/opt/myapp/
   ```

3. **服务器上运行**
   ```bash
   cd /opt/myapp
   chmod +x myapp
   ./myapp
   ```

### 用 systemd 管理服务

创建 `/etc/systemd/system/myapp.service`：

```ini
[Unit]
Description=My Go App
After=network.target

[Service]
Type=simple
User=www-data
WorkingDirectory=/opt/myapp
ExecStart=/opt/myapp/myapp
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target
```

```bash
sudo systemctl daemon-reload
sudo systemctl enable myapp
sudo systemctl start myapp
sudo systemctl status myapp
sudo journalctl -u myapp -f    # 查看日志
```

## 7. Docker 部署

### Dockerfile

```dockerfile
# 构建阶段
FROM golang:1.22-alpine AS builder

WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download

COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -o myapp main.go

# 运行阶段
FROM alpine:latest

WORKDIR /app
COPY --from=builder /app/myapp .
COPY --from=builder /app/config.yaml .

EXPOSE 8080
CMD ["./myapp"]
```

### 构建并运行

```bash
# 构建镜像
docker build -t myapp:latest .

# 运行容器
docker run -d \
  --name myapp \
  -p 8080:8080 \
  -e SERVER_PORT=8080 \
  -e DATABASE_HOST=host.docker.internal \
  myapp:latest

# 查看日志
docker logs -f myapp
```

### docker-compose.yml

```yaml
version: '3'
services:
  app:
    build: .
    ports:
      - "8080:8080"
    depends_on:
      - db
    environment:
      - DATABASE_HOST=db
      - DATABASE_PORT=3306
      - DATABASE_USERNAME=root
      - DATABASE_PASSWORD=secret
      - DATABASE_DBNAME=go_demo

  db:
    image: mysql:8
    environment:
      MYSQL_ROOT_PASSWORD: secret
      MYSQL_DATABASE: go_demo
    ports:
      - "3306:3306"
    volumes:
      - db_data:/var/lib/mysql

volumes:
  db_data:
```

```bash
docker-compose up -d
```
